cmake_minimum_required(VERSION 3.24)
project(wmma_linear_identity CXX)
include(${EXECUTORCH_ROOT}/tools/cmake/Utils.cmake)
find_package(executorch CONFIG REQUIRED COMPONENTS vulkan_backend)
executorch_target_link_options_shared_lib(vulkan_backend)
add_executable(wmma_linear_identity main.cpp ${EXECUTORCH_ROOT}/backends/vulkan/test/sarc_dev/utils.cpp)
target_include_directories(wmma_linear_identity PRIVATE ${EXECUTORCH_ROOT}/.. ${EXECUTORCH_ROOT}/backends/vulkan/test/sarc_dev ${EXECUTORCH_ROOT}/backends/vulkan/third-party/Vulkan-Headers/include ${EXECUTORCH_ROOT}/backends/vulkan/third-party/volk ${EXECUTORCH_ROOT}/backends/vulkan/third-party/VulkanMemoryAllocator)
target_link_libraries(wmma_linear_identity PRIVATE vulkan_backend executorch_core)
target_compile_options(wmma_linear_identity PRIVATE -fexceptions -DUSE_VULKAN_WRAPPER -DUSE_VULKAN_VOLK)
set_property(TARGET wmma_linear_identity PROPERTY CXX_STANDARD 17)
