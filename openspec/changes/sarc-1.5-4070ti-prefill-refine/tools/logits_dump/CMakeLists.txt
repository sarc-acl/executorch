cmake_minimum_required(VERSION 3.24)
project(logits_dump CXX)
set(CMAKE_CXX_STANDARD 17)
include(${EXECUTORCH_ROOT}/tools/cmake/Utils.cmake)
find_package(executorch CONFIG REQUIRED FIND_ROOT_PATH_BOTH)
executorch_target_link_options_shared_lib(executorch)
set(libs executorch extension_module extension_tensor extension_data_loader)
foreach(t optimized_native_cpu_ops_lib quantized_ops_lib custom_ops vulkan_backend)
  if(TARGET ${t})
    list(APPEND libs ${t})
    executorch_target_link_options_shared_lib(${t})
  endif()
endforeach()
foreach(t optimized_kernels portable_kernels quantized_kernels cpublas eigen_blas extension_threadpool pthreadpool cpuinfo)
  if(TARGET ${t})
    list(APPEND libs ${t})
  endif()
endforeach()
add_executable(logits_dump main.cpp)
target_link_libraries(logits_dump PRIVATE ${libs})
